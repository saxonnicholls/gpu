151class opencl_quant_kernels {
152public:
156 [[nodiscard]] static result<std::unique_ptr<opencl_quant_kernels>, opencl_failure> create(const opencl_context& context)
157 {
158 std::unique_ptr<opencl_quant_kernels> k(new opencl_quant_kernels(context));
159 cl_device_fp_config fp = 0;
160 clGetDeviceInfo(context.native_device(), CL_DEVICE_SINGLE_FP_CONFIG, sizeof(fp), &fp, nullptr);
161 k->exact_q8_ = (fp & CL_FP_CORRECTLY_ROUNDED_DIVIDE_SQRT) != 0;
162 const std::string source = detail::quant::program(detail::quant::opencl_prelude);
163 const char* text = source.c_str();
164 cl_int code = CL_SUCCESS;
165 k->program_ = clCreateProgramWithSource(context.native_context(), 1, &text, nullptr, &code);
166 if (code != CL_SUCCESS)
167 return err(opencl_failure{opencl_error::call_failed, code, {}});
168 cl_device_id device = context.native_device();
169 const char* options = k->exact_q8_ ? "-cl-fp32-correctly-rounded-divide-sqrt" : "";
170 if (const cl_int built = clBuildProgram(k->program_, 1, &device, options, nullptr, nullptr); built != CL_SUCCESS) {
171 std::size_t size = 0;
172 clGetProgramBuildInfo(k->program_, device, CL_PROGRAM_BUILD_LOG, 0, nullptr, &size);
173 std::string log(size, '\0');
174 clGetProgramBuildInfo(k->program_, device, CL_PROGRAM_BUILD_LOG, size, log.data(), nullptr);
175 return err(opencl_failure{opencl_error::build_failed, built, std::move(log)});
176 }
177 const auto make = [&](const std::string& name, cl_kernel& slot) {
178 slot = clCreateKernel(k->program_, name.c_str(), &code);
179 return code == CL_SUCCESS;
180 };
181 for (std::size_t f = 0; f < detail::quant::format_count; ++f) {
182 if (!detail::quant::available(detail::quant::format(f)))
183 continue;
184 const std::string n(detail::quant::names[f]);
185 if (!make("qmv_" + n, k->mv_[f]) || !make("qmm_" + n, k->mm_[f]) || !make("qmmt_" + n, k->mmt_[f])
186 || ((detail::quant::has_q8_0_dot(detail::quant::format(f)) || detail::quant::has_q8_1_dot(detail::quant::format(f)))
187 && !make("qdot_" + n, k->dot_[f]))
188 || (detail::quant::has_q8_k_dot(detail::quant::format(f)) && !make("qkdot_" + n, k->dot_[f]))
189 || (detail::quant::has_absmax_dot(detail::quant::format(f)) && !make("qdot_" + n, k->dot_[f])))
190 return err(opencl_failure{opencl_error::call_failed, code, {}});
191 }
192 if (!make("quantize_q8", k->quantize_) || !make("quantize_q8_1", k->quantize_1_) || !make("quantize_q8k", k->quantize_k_) || !make("quantize_absmax", k->quantize_a_))
193 return err(opencl_failure{opencl_error::call_failed, code, {}});
194 return ok(std::move(k));
195 }

230 template <typename W>
231 requires detail::quant::weights<W>
232 [[nodiscard]] opencl_quant_matrix upload(const W& w) const
233 {
234 const detail::quant::parts p = detail::quant::traits<W>::of(w);
235 if (!detail::quant::valid(p))
236 throw std::invalid_argument("snicholls::gpu::opencl_quant_kernels::upload: the weights' size does not match their shape and format");
237 auto s = std::make_shared<opencl_quant_matrix::storage>();
238 s->fmt = p.fmt;
239 s->mean = p.mean;
240 s->bytes = p.weights.size() + p.block_scales.size() + p.group_scales.size();
241 s->w = make_buffer(p.weights);
242 s->sb = p.block_scales.empty() ? nullptr : make_buffer(p.block_scales);
243 s->gs = p.group_scales.empty() ? nullptr : make_buffer(p.group_scales);
244 opencl_quant_matrix m;
245 m.rows = p.rows;
246 m.cols = p.cols;
247 m.s_ = std::move(s);
248 return m;
249 }

254 void forward(const opencl_quant_matrix& w, std::span<const float> xs, std::span<float> ys, std::span<const float> bias) const
255 {
256 const std::size_t n = rows_of(w, xs, w.cols, ys, w.rows, bias);
257 if (n == 0 || w.rows == 0)
258 return;
259 const cl_context ctx = context_->native_context();
260 detail::buffer db(ctx, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR, bias.size_bytes(), const_cast<float*>(bias.data()));
261 detail::buffer dx(ctx, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR, xs.size_bytes(), const_cast<float*>(xs.data()));
262 detail::buffer dy(ctx, CL_MEM_WRITE_ONLY, ys.size_bytes(), nullptr);
263 forward_buffers(w, dx.mem, dy.mem, db.mem, !bias.empty(), n);
264 detail::check(clEnqueueReadBuffer(context_->native_queue(), dy.mem, CL_TRUE, 0, ys.size_bytes(), ys.data(), 0, nullptr, nullptr),
265 "clEnqueueReadBuffer");
266 }

273 void forward_buffers(const opencl_quant_matrix& w, cl_mem x, cl_mem y, cl_mem bias, bool has_bias, std::size_t n) const
274 {
275 if (n == 0 || w.rows == 0)
276 return;
277 const auto& s = *w.s_;
278 const std::size_t f = std::size_t(s.fmt);
279 const cl_int hb = has_bias ? 1 : 0;
280 if (n <= matvec_max) {
281 // Blocks of mv_tokens rows; after the first, through two buffers of one block (copies at any offset: a
282 // sub-buffer's origin must be aligned, to 4 kB on some devices), kept until the queue has run them
283 const cl_command_queue q = context_->native_queue();
284 const std::size_t block = std::min(detail::quant::mv_tokens, n);
285 cl_mem bx = nullptr, by = nullptr;
286 for (std::size_t t0 = 0; t0 < n; t0 += detail::quant::mv_tokens) {
287 const cl_ulong count = cl_ulong(std::min(detail::quant::mv_tokens, n - t0));
288 cl_mem px = x, py = y;
289 if (t0 > 0) {
290 cl_int code = CL_SUCCESS;
291 if (!bx) {
292 bx = clCreateBuffer(context_->native_context(), CL_MEM_READ_WRITE, block * w.cols * sizeof(float), nullptr, &code);
293 detail::check(code, "clCreateBuffer");
294 by = clCreateBuffer(context_->native_context(), CL_MEM_READ_WRITE, block * w.rows * sizeof(float), nullptr, &code);
295 detail::check(code, "clCreateBuffer");
296 }
297 detail::check(clEnqueueCopyBuffer(q, x, bx, t0 * w.cols * sizeof(float), 0, std::size_t(count) * w.cols * sizeof(float), 0, nullptr,
298 nullptr),
299 "clEnqueueCopyBuffer");
300 px = bx, py = by;
301 }
302 const cl_ulong rows = w.rows;
303 {
304 const std::lock_guard lock(mutex_);
305 cl_kernel k = mv_[f];
306 cl_uint i = weight_args(k, s, w.cols);
307 set(k, i, px), set(k, i, count), set(k, i, rows), set(k, i, bias), set(k, i, hb), set(k, i, py);
308 const std::size_t global = (w.rows + detail::quant::mv_rows - 1) / detail::quant::mv_rows * 64, local = 64;
309 detail::check(clEnqueueNDRangeKernel(q, k, 1, nullptr, &global, &local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
310 }
311 if (t0 > 0)
312 detail::check(clEnqueueCopyBuffer(q, by, y, 0, t0 * w.rows * sizeof(float), std::size_t(count) * w.rows * sizeof(float), 0, nullptr,
313 nullptr),
314 "clEnqueueCopyBuffer");
315 }
316 if (bx) {
317 clFinish(q);
318 clReleaseMemObject(bx), clReleaseMemObject(by);
319 }
320 return;
321 }
322 tiled(mm_[f], s, w.cols, n, w.rows, w.cols, x, bias, hb, 0.0f, y);
323 }

345 void forward_q8(const opencl_quant_matrix& w, std::span<const float> xs, std::span<float> ys, std::span<const float> bias) const
346 {
347 const std::size_t n = rows_of(w, xs, w.cols, ys, w.rows, bias);
348 if (!detail::quant::has_q8_dot(w.s_->fmt))
349 throw std::invalid_argument("snicholls::gpu: 8-bit activations need weights of a format with core's integer dot product");
350 if (n == 0 || w.rows == 0)
351 return;
352 const auto ctx = context_->native_context();
353 const auto queue = context_->native_queue();
354 if (detail::quant::has_absmax_dot(w.s_->fmt)) {
355 forward_absmax(w, xs, ys, bias, n);
356 return;
357 }
358 const bool k_quant = detail::quant::has_q8_k_dot(w.s_->fmt), q8_1 = detail::quant::has_q8_1_dot(w.s_->fmt);
359 const cl_ulong blocks = n * w.cols / (k_quant ? 256 : 32);
360 cl_kernel quantize = k_quant ? quantize_k_ : q8_1 ? quantize_1_ : quantize_;
361 const std::size_t block_bytes = k_quant ? detail::quant::q8k_bytes : q8_1 ? detail::quant::q8_1_bytes : detail::quant::q8_bytes;
362 detail::buffer dx(ctx, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR, xs.size_bytes(), const_cast<float*>(xs.data()));
363 detail::buffer dq(ctx, CL_MEM_READ_WRITE, std::size_t(blocks) * block_bytes, nullptr);
364 detail::buffer dy(ctx, CL_MEM_WRITE_ONLY, ys.size_bytes(), nullptr);
365 detail::buffer db(ctx, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR, bias.size_bytes(), const_cast<float*>(bias.data()));
366 const cl_int has_bias = bias.empty() ? 0 : 1;
367 const cl_ulong cols = w.cols, rows = w.rows, t0 = 0;
368 const std::lock_guard lock(mutex_);
369 if (blocks > 0) {
370 cl_uint i = 0;
371 set(quantize, i, dx.mem), set(quantize, i, blocks), set(quantize, i, dq.mem);
372 const std::size_t global = (std::size_t(blocks) + 63) / 64 * 64, local = 64;
373 detail::check(clEnqueueNDRangeKernel(queue, quantize, 1, nullptr, &global, &local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
374 }
375 cl_kernel k = dot_[std::size_t(w.s_->fmt)];
376 cl_uint i = 0;
377 set(k, i, w.s_->w), set(k, i, cols), set(k, i, dq.mem), set(k, i, rows), set(k, i, t0), set(k, i, db.mem), set(k, i, has_bias), set(k, i, dy.mem);
378 const std::size_t global[2] = {w.rows * 64, n}, local[2] = {64, 1};
379 detail::check(clEnqueueNDRangeKernel(queue, k, 2, nullptr, global, local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
380 detail::check(clEnqueueReadBuffer(queue, dy.mem, CL_TRUE, 0, ys.size_bytes(), ys.data(), 0, nullptr, nullptr), "clEnqueueReadBuffer");
381 }

386 // BitLinear: the tokens' absmax int8 activations, then (acc / s) γ per output
387 void forward_absmax(const opencl_quant_matrix& w, std::span<const float> xs, std::span<float> ys, std::span<const float> bias, std::size_t n) const
388 {
389 const auto ctx = context_->native_context();
390 const auto queue = context_->native_queue();
391 detail::buffer dx(ctx, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR, xs.size_bytes(), const_cast<float*>(xs.data()));
392 detail::buffer dq(ctx, CL_MEM_READ_WRITE, n * (w.cols + 4), nullptr);
393 detail::buffer dy(ctx, CL_MEM_WRITE_ONLY, ys.size_bytes(), nullptr);
394 detail::buffer db(ctx, CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR, bias.size_bytes(), const_cast<float*>(bias.data()));
395 const cl_int has_bias = bias.empty() ? 0 : 1;
396 const cl_ulong cols = w.cols, rows = w.rows, t0 = 0;
397 const float gamma = w.s_->mean;
398 const std::lock_guard lock(mutex_);
399 cl_uint i = 0;
400 set(quantize_a_, i, dx.mem), set(quantize_a_, i, cols), set(quantize_a_, i, dq.mem);
401 const std::size_t global = n * 64, local = 64;
402 detail::check(clEnqueueNDRangeKernel(queue, quantize_a_, 1, nullptr, &global, &local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
403 cl_kernel k = dot_[std::size_t(w.s_->fmt)];
404 i = 0;
405 set(k, i, w.s_->w), set(k, i, cols), set(k, i, dq.mem), set(k, i, rows), set(k, i, t0), set(k, i, gamma), set(k, i, db.mem), set(k, i, has_bias),
406 set(k, i, dy.mem);
407 const std::size_t g2[2] = {w.rows * 64, n}, l2[2] = {64, 1};
408 detail::check(clEnqueueNDRangeKernel(queue, k, 2, nullptr, g2, l2, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
409 detail::check(clEnqueueReadBuffer(queue, dy.mem, CL_TRUE, 0, ys.size_bytes(), ys.data(), 0, nullptr, nullptr), "clEnqueueReadBuffer");
410 }

428 // The weight arguments every format's kernels start with (WARGS); a format without scales reads none, so its
429 // weights stand in for them
430 static cl_uint weight_args(cl_kernel k, const opencl_quant_matrix::storage& s, std::size_t cols)
431 {
432 cl_uint i = 0;
433 const cl_ulong c = cols;
434 set(k, i, s.w), set(k, i, s.sb ? s.sb : s.w), set(k, i, s.gs ? s.gs : s.w), set(k, i, s.mean), set(k, i, c);
435 return i;
436 }

438 void tiled(cl_kernel k, const opencl_quant_matrix::storage& s, std::size_t cols, std::size_t m, std::size_t nn, std::size_t kk, cl_mem a,
439 cl_mem bias, cl_int has_bias, float beta, cl_mem c) const
440 {
441 const cl_ulong mm = m, n = nn, kd = kk;
442 const std::lock_guard lock(mutex_);
443 cl_uint i = weight_args(k, s, cols);
444 set(k, i, mm), set(k, i, n), set(k, i, kd), set(k, i, a), set(k, i, bias), set(k, i, has_bias), set(k, i, beta), set(k, i, c);
445 const std::size_t global[2] = {(nn + 63) / 64 * 16, (m + 63) / 64 * 16}, local[2] = {16, 16};
446 detail::check(clEnqueueNDRangeKernel(context_->native_queue(), k, 2, nullptr, global, local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
447 }

449 // The rows of a product, after checking every size against the matrix
450 static std::size_t rows_of(const opencl_quant_matrix& w, std::span<const float> in, std::size_t in_width, std::span<const float> out,
451 std::size_t out_width, std::span<const float> bias)
452 {
453 if (!w.s_)
454 throw std::invalid_argument("snicholls::gpu: an opencl_quant_matrix without weights");
455 const std::size_t n = in_width == 0 ? (out_width == 0 ? 0 : out.size() / out_width) : in.size() / in_width;
456 if ((in_width != 0 && in.size() % in_width != 0) || out.size() != n * out_width || (!bias.empty() && bias.size() != w.rows))
457 throw std::invalid_argument("snicholls::gpu: the activations' sizes do not fit the weights");
458 return n;
459 }
