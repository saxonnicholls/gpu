8// The quantised-weight kernels' source, written once for OpenCL C 1.2 and for CUDA C++ (NVRTC): a prelude per
9// language maps a handful of macros (KERNEL, GLOBAL, LOCAL, BARRIER, the thread indices, the bit casts), and the
10// formats and kernels below use only those. Not public API: opencl_quant.hpp and cuda_quant.hpp build their programs
11// from it.
12//
13// A format is one device function, `get_<format>(WARGS, r, c)`: the weight at row r, column c, dequantised exactly as
14// core's host code does it (ml/quantize.hpp, ml/nf4.hpp, ml/minifloat.hpp), so a product that adds the same products in
15// the same order gives core's bits. Its host side is `traits<W>` below. Adding a format is those two and its entry in
16// `formats`.

212// The weights' bytes a storage of this shape must have, and whether the shape is one the format allows (a block never
213// crosses a row): the overloads check both before a launch
214inline bool valid(const parts& p) noexcept
215{
216 const std::size_t n = p.rows * p.cols;
217 switch (p.fmt) {
218 case format::q8_0: return p.cols % 32 == 0 && p.weights.size() == n / 32 * sizeof(ml::block_q8_0);
219 case format::q4_0: return p.cols % 32 == 0 && p.weights.size() == n / 32 * sizeof(ml::block_q4_0);
220 case format::mxfp4: return p.cols % 32 == 0 && p.weights.size() == n / 32 * sizeof(ml::block_mxfp4);
221 case format::nf4:
222 return p.cols % ml::nf4_block_size == 0 && p.weights.size() == ml::nf4_code_bytes(n) && p.block_scales.size() == ml::nf4_block_count(n)
223 && p.group_scales.size() == ml::nf4_group_count(n) * sizeof(float);
224 default: {
225 const std::size_t e = block_elements[std::size_t(p.fmt)];
226 return p.cols % e == 0 && p.weights.size() == n / e * block_bytes[std::size_t(p.fmt)];
227 }
228 }

288// A minifloat's exact value (ml/minifloat.hpp to_float): E exponent bits, M mantissa bits, kind 0 IEEE (the all-ones
289// exponent infinite or NaN), 1 finite with one NaN (E4M3)
290DEVICE float mf_decode(uint b, int E, int M, int kind)
291{
292 const int bias = (1 << (E - 1)) - 1;
293 const uint sign = (b >> (E + M)) & 1u, e = (b >> M) & ((1u << E) - 1u), m = b & ((1u << M) - 1u);
294 float v;
295 if (kind == 0 && e == (1u << E) - 1u) v = m == 0u ? AS_FLOAT(0x7f800000u) : AS_FLOAT(0x7fc00000u);
296 else if (kind == 1 && e == (1u << E) - 1u && m == (1u << M) - 1u) v = AS_FLOAT(0x7fc00000u);
297 else if (e == 0u) v = (float)m * AS_FLOAT((uint)(1 - bias - M + 127) << 23); // exact: m below 2^M times a power of two
298 else v = AS_FLOAT((((uint)((int)e - bias + 127)) << 23) | (m << (23 - M)));
299 return sign ? -v : v;
300}
301DEVICE float half_at(GLOBAL const uchar* p) { return mf_decode((uint)p[0] | ((uint)p[1] << 8), 5, 10, 0); }
302// 2^(e - 128), half the MX scale (ml/quantize.hpp e8m0_half)
303DEVICE float e8m0_half(uint e) { return AS_FLOAT(e < 2u ? 0x00200000u << e : (e - 1u) << 23); }
304// E2M1 doubled (ggml's kvalues_mxfp4) and NormalFloat4 (ml/nf4.hpp nf4_values)

310// Q8_0: 34-byte blocks, a float16 scale then 32 signed bytes; x = q d
311DEVICE float get_q8_0(WARGS, u64 r, u64 c)
312{
313 GLOBAL const uchar* b = w + (r * (cols / 32) + c / 32) * 34;
314 return (float)(i8)b[2 + c % 32] * half_at(b);
315}
316// Q4_0: 18-byte blocks, a float16 scale then element j in the low nibble of byte j, j + 16 in the high; x = (q - 8) d
317DEVICE float get_q4_0(WARGS, u64 r, u64 c)
318{
319 GLOBAL const uchar* b = w + (r * (cols / 32) + c / 32) * 18;
320 const uint j = (uint)(c % 32), q = b[2 + (j & 15u)];
321 return (float)((int)(j < 16u ? q & 15u : q >> 4) - 8) * half_at(b);
322}
323// MXFP4: 17-byte blocks, an E8M0 exponent then E2M1 codes packed as Q4_0's; x = 2^(e - 128) 2v
324DEVICE float get_mxfp4(WARGS, u64 r, u64 c)
325{
326 GLOBAL const uchar* b = w + (r * (cols / 32) + c / 32) * 17;
327 const uint j = (uint)(c % 32), q = b[1 + (j & 15u)];
328 return e8m0_half(b[0]) * (float)fp4_doubled[j < 16u ? q & 15u : q >> 4];
329}
330// NF4: two codes a byte, the first in the high nibble; a block of 64 scaled by e4m3(sb) s1 + mean, s1 its group's of 256
331DEVICE float get_nf4(WARGS, u64 r, u64 c)
332{
333 const u64 i = r * cols + c, blk = i / 64;
334 const uint q = w[i / 2], code = (i % 2 == 0) ? q >> 4 : q & 15u;
335 const float s = mf_decode(sb[blk], 4, 3, 1) * gs[blk / 256];
336 const float scale = s + mean;
337 return nf4_values[code] * scale;
338}

345// Eight consecutive weights of one row from column c0 (c0 a multiple of 8, c0 + 8 at most cols), the same values as
346// get_<format>: the matrix-vector kernels' unit, decoding a block's scale once
347DEVICE void get8_q8_0(WARGS, u64 r, u64 c0, float* v)
348{
349 GLOBAL const uchar* b = w + (r * (cols / 32) + c0 / 32) * 34;
