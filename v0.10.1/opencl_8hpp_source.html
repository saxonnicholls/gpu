137 // The kernels: C = alpha A B + beta C over strided views, 16 x 16 tiles, each output summed in index order with
138 // contraction off, so the result is linalg::reference's
139 inline constexpr const char* gemm_source = R"CLC(
140#pragma OPENCL FP_CONTRACT OFF
141// 64 x 64 tiles of C, 16 deep in k; a work-group of 16 x 16 items, each computing 4 x 4 outputs, every output summed
142// in index order. A tile of an operand is loaded with consecutive items along its unit stride (row-major or
143// column-major), so the loads coalesce whichever way the operand is laid out.
144#define TSM 64
145#define TSN 64
146#define TSK 16
147#define WPT 4
148#define RTS 16
149#define GEMM(NAME, T) \
150__kernel __attribute__((reqd_work_group_size(RTS, RTS, 1))) \
151void NAME(const long m, const long n, const long k, const T alpha, \
152 __global const T* a, const long sa0, const long sa1, __global const T* b, const long sb0, const long sb1, \
153 const T beta, __global T* c, const long sc0, const long sc1, const long oa, const long ob, const long oc) \
154{ \
155 a += oa, b += ob, c += oc; \
156 __local T as[TSK][TSM]; \
157 __local T bs[TSK][TSN]; \
158 const int tc = get_local_id(0), tr = get_local_id(1), tid = tr * RTS + tc; \
159 const long col0 = (long)get_group_id(0) * TSN, row0 = (long)get_group_id(1) * TSM; \
160 const bool a_by_rows = sa1 == 1 || sa0 != 1, b_by_rows = sb1 == 1 || sb0 != 1; \
161 T acc[WPT][WPT]; \
162 for (int i = 0; i < WPT; ++i) \
163 for (int j = 0; j < WPT; ++j) \
164 acc[i][j] = (T)0; \
165 for (long t = 0; t < k; t += TSK) { \
166 for (int l = 0; l < (TSM * TSK) / (RTS * RTS); ++l) { \
167 const int e = tid + l * RTS * RTS; \
168 const int ar = a_by_rows ? e / TSK : e % TSM, ap = a_by_rows ? e % TSK : e / TSM; \
169 const long gr = row0 + ar, gp = t + ap; \
170 as[ap][ar] = (gr < m && gp < k) ? a[gr * sa0 + gp * sa1] : (T)0; \
171 const int bp = b_by_rows ? e / TSN : e % TSK, bc = b_by_rows ? e % TSN : e / TSK; \
172 const long gq = t + bp, gc = col0 + bc; \
173 bs[bp][bc] = (gq < k && gc < n) ? b[gq * sb0 + gc * sb1] : (T)0; \
174 } \
175 barrier(CLK_LOCAL_MEM_FENCE); \
176 const int last = k - t < TSK ? (int)(k - t) : TSK; \
177 for (int p = 0; p < last; ++p) { \
178 T ar[WPT], br[WPT]; \
179 for (int i = 0; i < WPT; ++i) \
180 ar[i] = as[p][tr + i * RTS]; \
181 for (int j = 0; j < WPT; ++j) \
182 br[j] = bs[p][tc + j * RTS]; \
183 for (int i = 0; i < WPT; ++i) \
184 for (int j = 0; j < WPT; ++j) \
185 acc[i][j] += ar[i] * br[j]; \
186 } \
187 barrier(CLK_LOCAL_MEM_FENCE); \
188 } \
189 for (int i = 0; i < WPT; ++i) \
190 for (int j = 0; j < WPT; ++j) { \
191 const long row = row0 + tr + i * RTS, col = col0 + tc + j * RTS; \
192 if (row < m && col < n) { \
193 __global T* out = c + row * sc0 + col * sc1; \
194 const T s = acc[i][j]; \
195 *out = beta == (T)0 ? alpha * s : alpha * s + beta * *out; \
196 } \
197 } \
198}
199GEMM(gemm_float, float)
200#ifdef SNICHOLLS_GPU_FP64
201#pragma OPENCL EXTENSION cl_khr_fp64 : enable
202GEMM(gemm_double, double)
203#endif
204)CLC";

239public:
243 [[nodiscard]] static result<std::unique_ptr<opencl_context>, opencl_failure> create(device_choice choice = device_choice::any)
244 {
245 cl_uint platforms = 0;
246 if (clGetPlatformIDs(0, nullptr, &platforms) != CL_SUCCESS || platforms == 0)
247 return err(opencl_failure{opencl_error::no_platform, CL_SUCCESS, {}});
248 std::vector<cl_platform_id> ids(platforms);
249 clGetPlatformIDs(platforms, ids.data(), nullptr);
250 cl_device_id device = nullptr;
251 for (const cl_device_type type : {cl_device_type(CL_DEVICE_TYPE_GPU), cl_device_type(CL_DEVICE_TYPE_ALL)}) {
252 if (type == CL_DEVICE_TYPE_ALL && choice == device_choice::gpu)
253 break;
254 for (const cl_platform_id p : ids)
255 if (!device && clGetDeviceIDs(p, type, 1, &device, nullptr) != CL_SUCCESS)
256 device = nullptr;
257 if (device)
258 break;
259 }
260 if (!device)
261 return err(opencl_failure{opencl_error::no_device, CL_SUCCESS, {}});
262 std::unique_ptr<opencl_context> c(new opencl_context(device));
263 cl_int code = CL_SUCCESS;
264 c->context_ = clCreateContext(nullptr, 1, &device, nullptr, nullptr, &code);
265 if (code != CL_SUCCESS)
266 return err(opencl_failure{opencl_error::call_failed, code, {}});
267 c->queue_ = clCreateCommandQueue(c->context_, device, 0, &code);
268 if (code != CL_SUCCESS)
269 return err(opencl_failure{opencl_error::call_failed, code, {}});
270 const char* source = detail::gemm_source; // macOS takes const char**, others const char* const*
271 c->program_ = clCreateProgramWithSource(c->context_, 1, &source, nullptr, &code);
272 if (code != CL_SUCCESS)
273 return err(opencl_failure{opencl_error::call_failed, code, {}});
274 const char* options = c->fp64_ ? "-DSNICHOLLS_GPU_FP64" : "";
275 if (const cl_int built = clBuildProgram(c->program_, 1, &device, options, nullptr, nullptr); built != CL_SUCCESS) {
276 std::size_t size = 0;
277 clGetProgramBuildInfo(c->program_, device, CL_PROGRAM_BUILD_LOG, 0, nullptr, &size);
278 std::string log(size, '\0');
279 clGetProgramBuildInfo(c->program_, device, CL_PROGRAM_BUILD_LOG, size, log.data(), nullptr);
280 return err(opencl_failure{opencl_error::build_failed, built, std::move(log)});
281 }
282 c->gemm_float_ = clCreateKernel(c->program_, "gemm_float", &code);
283 if (code != CL_SUCCESS)
284 return err(opencl_failure{opencl_error::call_failed, code, {}});
285 if (c->fp64_) {
286 c->gemm_double_ = clCreateKernel(c->program_, "gemm_double", &code);
287 if (code != CL_SUCCESS)
288 return err(opencl_failure{opencl_error::call_failed, code, {}});
289 }
290 return ok(std::move(c));
291 }

321 template <typename T>
322 void gemm(T alpha, linalg::matrix_view<const T> a, linalg::matrix_view<const T> b, T beta, linalg::matrix_view<T> c) const
323 {
324 const std::size_t m = c.extent(0), n = c.extent(1), k = a.extent(1);
325 if (m == 0 || n == 0)
326 return;
327 const std::size_t sa = a.mapping().required_span_size(), sb = b.mapping().required_span_size(),
328 sc = c.mapping().required_span_size();
329 const cl_mem_flags in = CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR;
330 // An empty inner dimension still writes alpha * 0 (+ beta c): the kernel's loop runs no times
331 detail::buffer da(context_, in, sa * sizeof(T), const_cast<T*>(a.data_handle()));
332 detail::buffer db(context_, in, sb * sizeof(T), const_cast<T*>(b.data_handle()));
333 // C goes up only if the kernel reads it (beta != 0) or its span has gaps the copy back must keep
334 const bool upload = beta != T(0) || !c.mapping().is_exhaustive();
335 detail::buffer dc(context_, upload ? CL_MEM_READ_WRITE | CL_MEM_COPY_HOST_PTR : CL_MEM_WRITE_ONLY, sc * sizeof(T),
336 upload ? c.data_handle() : nullptr);
337 gemm_buffers(alpha, m, n, k, da.mem, 0, a.stride(0), a.stride(1), db.mem, 0, b.stride(0), b.stride(1), beta, dc.mem, 0, c.stride(0),
338 c.stride(1));
339 detail::check(clEnqueueReadBuffer(queue_, dc.mem, CL_TRUE, 0, sc * sizeof(T), c.data_handle(), 0, nullptr, nullptr),
340 "clEnqueueReadBuffer");
341 }

350 template <typename T>
351 void gemm_buffers(T alpha, std::size_t m, std::size_t n, std::size_t k, cl_mem a, std::size_t oa, std::size_t sa0, std::size_t sa1, cl_mem b,
352 std::size_t ob, std::size_t sb0, std::size_t sb1, T beta, cl_mem c, std::size_t oc, std::size_t sc0, std::size_t sc1) const
353 {
354 if (m == 0 || n == 0)
355 return;
356 const cl_long args[] = {cl_long(m), cl_long(n), cl_long(k), cl_long(sa0), cl_long(sa1), cl_long(sb0), cl_long(sb1), cl_long(sc0),
357 cl_long(sc1), cl_long(oa), cl_long(ob), cl_long(oc)};
358 const std::lock_guard lock(mutex_);
359 cl_kernel kernel = std::is_same_v<T, float> ? gemm_float_ : gemm_double_;
360 cl_uint i = 0;
361 auto set = [&](std::size_t size, const void* value) { detail::check(clSetKernelArg(kernel, i++, size, value), "clSetKernelArg"); };
362 set(sizeof(cl_long), &args[0]), set(sizeof(cl_long), &args[1]), set(sizeof(cl_long), &args[2]), set(sizeof(T), &alpha);
363 set(sizeof(cl_mem), &a), set(sizeof(cl_long), &args[3]), set(sizeof(cl_long), &args[4]);
364 set(sizeof(cl_mem), &b), set(sizeof(cl_long), &args[5]), set(sizeof(cl_long), &args[6]);
365 set(sizeof(T), &beta), set(sizeof(cl_mem), &c), set(sizeof(cl_long), &args[7]), set(sizeof(cl_long), &args[8]);
366 set(sizeof(cl_long), &args[9]), set(sizeof(cl_long), &args[10]), set(sizeof(cl_long), &args[11]);
367 const std::size_t global[2] = {(n + 63) / 64 * 16, (m + 63) / 64 * 16}, local[2] = {16, 16}; // 64 x 64 per group
368 detail::check(clEnqueueNDRangeKernel(queue_, kernel, 2, nullptr, global, local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
369 }
