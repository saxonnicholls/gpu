118 [[nodiscard]] static std::unique_ptr<opencl_arena> create(const opencl_context& context, std::size_t reserve = std::size_t{64} << 30,
119 std::size_t chunk = std::size_t{1} << 30)
120 {
121 cl_ulong max_alloc = 0;
122 clGetDeviceInfo(context.native_device(), CL_DEVICE_MAX_MEM_ALLOC_SIZE, sizeof(max_alloc), &max_alloc, nullptr);
123 void* base = ::mmap(nullptr, reserve, PROT_NONE, MAP_PRIVATE | MAP_ANON, -1, 0);
124 if (base == MAP_FAILED)
125 throw std::bad_alloc();
126 return std::unique_ptr<opencl_arena>(new opencl_arena(context, static_cast<std::byte*>(base), reserve,
127 std::min<std::size_t>(chunk, std::size_t(max_alloc)), std::size_t(max_alloc)));
128 }

144 template <typename T>
145 [[nodiscard]] std::span<T> allocate(std::size_t n)
146 {
147 static_assert(std::is_trivially_copyable_v<T>);
148 return allocate_bytes<T>(n, round(std::max<std::size_t>(n * sizeof(T), 1)));

151private:
152 template <typename T>
153 [[nodiscard]] std::span<T> allocate_bytes(std::size_t n, std::size_t bytes)
154 {
155 const std::lock_guard lock(mutex_);
156 for (auto it = free_.begin(); it != free_.end(); ++it) // first fit among the freed blocks
157 if (it->size >= bytes) {
158 const std::size_t at = it->start;
159 if (it->size == bytes)
160 free_.erase(it);
161 else
162 it->start += bytes, it->size -= bytes;
163 return {reinterpret_cast<T*>(base_ + at), n};
164 }
165 if (chunks_.empty() || round(top_) + bytes > chunks_.back().start + chunks_.back().size) {
166 const std::size_t start = chunks_.empty() ? 0 : round(chunks_.back().start + chunks_.back().size);
167 const std::size_t size = std::max(chunk_, round(bytes));
168 if (size > max_alloc_ || start + size > reserve_)
169 throw std::bad_alloc();
170 cl_int code = CL_SUCCESS;
171 cl_mem mem = clCreateBuffer(context_->native_context(), CL_MEM_READ_WRITE, size, nullptr, &code);
172 detail::check(code, "clCreateBuffer (the arena)");
173 chunks_.push_back({mem, start, size});
174 top_ = start;
175 }
176 const std::size_t at = round(top_);
177 top_ = at + bytes;
178 return {reinterpret_cast<T*>(base_ + at), n};
179 }

184 template <typename T>
185 void deallocate(std::span<T> s)
186 {
187 if (!owns(s.data()))
188 return;
189 const std::size_t at = std::size_t(reinterpret_cast<const std::byte*>(s.data()) - base_), bytes = round(std::max<std::size_t>(s.size_bytes(), 1));
190 const std::lock_guard lock(mutex_);
191 auto it = free_.begin();
192 while (it != free_.end() && it->start < at)
193 ++it;
194 it = free_.insert(it, {at, bytes});
195 if (std::next(it) != free_.end() && it->start + it->size == std::next(it)->start && same_chunk(it->start, std::next(it)->start))
196 it->size += std::next(it)->size, free_.erase(std::next(it));
197 if (it != free_.begin() && std::prev(it)->start + std::prev(it)->size == it->start && same_chunk(std::prev(it)->start, it->start)) {
198 std::prev(it)->size += it->size;
199 free_.erase(it);
200 }
201 }

205 [[nodiscard]] location locate(const void* p) const
206 {
207 const auto* b = static_cast<const std::byte*>(p);
208 if (b < base_ || b >= base_ + reserve_)
209 throw std::invalid_argument("snicholls::gpu: a span that is not in the OpenCL arena");
210 const std::size_t at = std::size_t(b - base_);
211 const std::lock_guard lock(mutex_);
212 for (const auto& c : chunks_)
213 if (at >= c.start && at < c.start + c.size)
214 return {c.mem, at - c.start};
215 throw std::invalid_argument("snicholls::gpu: a span in the arena's reservation but in no buffer");
216 }
217
219 [[nodiscard]] bool owns(const void* p) const noexcept
220 {
221 const auto* b = static_cast<const std::byte*>(p);
222 return b >= base_ && b < base_ + reserve_;
223 }

254 void release(std::size_t m)
255 {
256 const std::lock_guard lock(mutex_);
257 top_ = std::min(top_, m);
258 std::erase_if(free_, [&](const block& b) { return b.start >= top_; });
259 while (!chunks_.empty() && chunks_.back().start > top_) {
260 clReleaseMemObject(chunks_.back().mem);
261 chunks_.pop_back();
262 }

307class opencl_device_kernels {
308public:
310 [[nodiscard]] static result<std::unique_ptr<opencl_device_kernels>, opencl_failure> create(const opencl_context& context)
311 {
312 std::unique_ptr<opencl_device_kernels> k(new opencl_device_kernels(context));
313 const std::string source = std::string(detail::quant::opencl_prelude) + detail::rows::math_opencl + detail::rows::source;
314 const char* text = source.c_str();
315 cl_int code = CL_SUCCESS;
316 k->program_ = clCreateProgramWithSource(context.native_context(), 1, &text, nullptr, &code);
317 if (code != CL_SUCCESS)
318 return err(opencl_failure{opencl_error::call_failed, code, {}});
319 cl_device_id device = context.native_device();
320 if (const cl_int built = clBuildProgram(k->program_, 1, &device, "", nullptr, nullptr); built != CL_SUCCESS) {
321 std::size_t size = 0;
322 clGetProgramBuildInfo(k->program_, device, CL_PROGRAM_BUILD_LOG, 0, nullptr, &size);
323 std::string log(size, '\0');
324 clGetProgramBuildInfo(k->program_, device, CL_PROGRAM_BUILD_LOG, size, log.data(), nullptr);
325 return err(opencl_failure{opencl_error::build_failed, built, std::move(log)});
326 }
327 for (std::size_t i = 0; i < kernel_count; ++i) {
328 k->kernels_[i] = clCreateKernel(k->program_, names[i], &code);
329 if (code != CL_SUCCESS)
330 return err(opencl_failure{opencl_error::call_failed, code, {}});
331 }
332 return ok(std::move(k));
333 }

363 template <typename... A>
364 void launch(id k, std::size_t groups, std::size_t local, const A&... args) const
365 {
366 if (groups == 0)
367 return;
368 const std::lock_guard lock(mutex_);
369 cl_kernel kernel = kernels_[std::size_t(k)];
370 cl_uint i = 0;
371 ((detail::check(clSetKernelArg(kernel, i++, sizeof(A), &args), "clSetKernelArg")), ...);
372 const std::size_t global = groups * local;
373 detail::check(clEnqueueNDRangeKernel(context_->native_queue(), kernel, 1, nullptr, &global, &local, 0, nullptr, nullptr),
374 "clEnqueueNDRangeKernel");
375 debug_sync(names[std::size_t(k)]);
376 }

398 void quant_product(const opencl_quant_kernels& quant, const opencl_quant_matrix& w, opencl_arena::location x, std::size_t xb,
399 opencl_arena::location y, std::size_t yb, opencl_arena::location bias, std::size_t bb, std::size_t n,
400 std::size_t rows_max = 512) const
401 {
402 const std::lock_guard lock(scratch_mutex_);
403 const cl_command_queue q = context_->native_queue();
404 const std::size_t step = rows_max == 0 || rows_max > n ? n : rows_max, xr = xb / n, yr = yb / n; // bytes a row
405 grow(scratch_[0], step * xr), grow(scratch_[1], step * yr), grow(scratch_[2], std::max<std::size_t>(bb, 4));
406 if (bias.mem)
407 detail::check(clEnqueueCopyBuffer(q, bias.mem, scratch_[2].mem, bias.offset, 0, bb, 0, nullptr, nullptr), "clEnqueueCopyBuffer");
408 for (std::size_t r = 0; r < n; r += step) { // a launch of at most step rows (the same results; no launch long enough for a watchdog)
409 const std::size_t m = std::min(step, n - r);
410 detail::check(clEnqueueCopyBuffer(q, x.mem, scratch_[0].mem, x.offset + r * xr, 0, m * xr, 0, nullptr, nullptr), "clEnqueueCopyBuffer");
411 quant.forward_buffers(w, scratch_[0].mem, scratch_[1].mem, scratch_[2].mem, bias.mem != nullptr, m);
412 debug_sync("quantised product");
413 detail::check(clEnqueueCopyBuffer(q, scratch_[1].mem, y.mem, 0, y.offset + r * yr, m * yr, 0, nullptr, nullptr), "clEnqueueCopyBuffer");
414 }
415 }

457 {

517 template <linalg::scalar T>
518 void gemm(T alpha, linalg::matrix_view<const T> a, linalg::matrix_view<const T> b, T beta, linalg::matrix_view<T> c) const
519 {
520 static_assert(std::is_same_v<T, float>, "snicholls::gpu: opencl_device_backend multiplies float");
521 if (c.extent(0) == 0 || c.extent(1) == 0)
522 return;
523 const auto la = arena->locate(a.data_handle()), lb = arena->locate(b.data_handle()), lc = arena->locate(c.data_handle());
524 // A few rows times a transposed row-major matrix (ml::linear_sequence's decode step): one item an output
525 if (c.extent(0) <= matvec_max && alpha == T(1) && beta == T(0) && a.stride(1) == 1 && b.stride(0) == 1 && b.stride(1) == a.extent(1)
526 && c.stride(1) == 1) {
527 kernels->launch(opencl_device_kernels::id::matvec, c.extent(1), 64, la.mem, cl_ulong(la.offset / sizeof(T)), lb.mem,
528 cl_ulong(lb.offset / sizeof(T)), lc.mem, cl_ulong(lc.offset / sizeof(T)), cl_ulong(c.extent(0)), cl_ulong(a.extent(1)),
529 cl_ulong(c.extent(1)), cl_ulong(a.stride(0)), cl_ulong(c.stride(0)));
530 return;
531 }
532 // At most `rows_max` rows of C a launch (rows are independent: the same results), so that no launch runs long
533 // enough for a driver's watchdog (macOS's kills a client whose kernel holds the GPU for seconds)
534 const std::size_t step = rows_max == 0 ? c.extent(0) : rows_max;
535 for (std::size_t r = 0; r < c.extent(0); r += step) {
536 const std::size_t m = std::min(step, c.extent(0) - r);
537 context->gemm_buffers(alpha, m, c.extent(1), a.extent(1), la.mem, la.offset / sizeof(T) + r * a.stride(0), a.stride(0), a.stride(1),
538 lb.mem, lb.offset / sizeof(T), b.stride(0), b.stride(1), beta, lc.mem, lc.offset / sizeof(T) + r * c.stride(0),
539 c.stride(0), c.stride(1));
540 }
541 kernels->debug_sync("product");
542 }

643void block_norm_rows(const opencl_device_backend& b, std::type_identity_t<std::span<const float>> x, std::type_identity_t<std::span<const float>> weight,
644 std::type_identity_t<std::span<const float>> bias, std::span<float> out)
645{
646 namespace d = detail::opencl_device;
647 if (x.empty())

674template <ml::transformer_config C, typename W, typename R>
675void attention_position_rows(const opencl_device_backend& b, const ml::attention_weights<W, float>& w, const ml::attention_activations<float>& act,
676 std::size_t n, const R& rope_src, std::size_t sequences = 1)
677{
678 if constexpr (C.qk_norm == ml::qk_norm_kind::none && C.rope == ml::rope_kind::none) {
679 (void)b, (void)w, (void)act, (void)n, (void)rope_src, (void)sequences;
680 } else {
681 namespace d = detail::opencl_device;
682 constexpr std::size_t half = C.head_dim / 2;
683 // RoPE's rows for positions 0 .. n - 1 of the source, up for this call (the source is the same for every sequence)
684 std::vector<float> cs(C.rope == ml::rope_kind::none ? 1 : n * half), sn(cs.size());
685 if constexpr (C.rope != ml::rope_kind::none)
686 for (std::size_t t = 0; t < n; ++t) {
687 const auto c = rope_src.cos_at(t), s = rope_src.sin_at(t);
688 std::copy(c.begin(), c.end(), cs.begin() + std::ptrdiff_t(t * half));
689 std::copy(s.begin(), s.end(), sn.begin() + std::ptrdiff_t(t * half));
690 }
691 const auto lock = b.kernels->rope_rows(cs, sn); // the rows in the kernels' own buffers, held until launched
692 cl_mem dcs = b.kernels->rope_cos(), dsn = b.kernels->rope_sin();
693 const bool raw = !act.q_raw.empty();
694 const auto pq = d::at(b, act.q), pk = d::at(b, act.k), pqr = d::at_or(b, act.q_raw, act.q), pkr = d::at_or(b, act.k_raw, act.k);
695 const auto pqn = d::at_or(b, w.q_norm, std::span<const float>(act.q)), pkn = d::at_or(b, w.k_norm, std::span<const float>(act.k));
696 const int qk = C.qk_norm == ml::qk_norm_kind::before_rope ? 1 : C.qk_norm == ml::qk_norm_kind::after_rope ? 2 : 0;
697 const std::size_t items = n * sequences * (C.n_heads + C.n_kv_heads);
698 b.kernels->launch(d::id::position, d::groups(items, 64), 64, pq.mem, pq.offset, pk.mem, pk.offset, pqr.mem, pqr.offset, pkr.mem, pkr.offset,
699 cl_int(raw ? 1 : 0), pqn.mem, pqn.offset, cl_int(w.q_norm.empty() ? 0 : 1), pkn.mem, pkn.offset,
700 cl_int(w.k_norm.empty() ? 0 : 1), dcs, dsn, cl_ulong(n), cl_ulong(sequences), cl_int(C.n_heads),
701 cl_int(C.n_kv_heads), cl_ulong(C.head_dim), cl_int(qk), cl_int(C.rope == ml::rope_kind::none ? 0 : 1),
702 cl_int(C.layout() == ml::rope_layout::half ? 1 : 0), ml::norm_eps_v<C, float>);
703
704 }

819 inline float reduce(const opencl_device_backend& b, std::span<const float> x, int squares)
820 {
821 float out = 0;
822 if (x.empty())
823 return out;
824 cl_int code = CL_SUCCESS;
825 cl_mem result = clCreateBuffer(b.context->native_context(), CL_MEM_WRITE_ONLY, sizeof(float), nullptr, &code);
826 check(code, "clCreateBuffer");
827 const auto px = at(b, x);
828 b.kernels->launch(id::sum, 1, 256, px.mem, px.offset, cl_ulong(x.size()), cl_int(squares), result);
829 code = clEnqueueReadBuffer(b.context->native_queue(), result, CL_TRUE, 0, sizeof(float), &out, 0, nullptr, nullptr);
830 clReleaseMemObject(result);
831 check(code, "clEnqueueReadBuffer");
832 return out;
833 }
