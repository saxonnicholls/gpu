127 // Whether CUDA knows the pointer as device (or managed) memory
128 inline bool on_device(const void* p) noexcept
129 {
130 if (p == nullptr)
131 return false;
132 cudaPointerAttributes a{};
133 if (cudaPointerGetAttributes(&a, p) != cudaSuccess) {
134 (void)cudaGetLastError(); // a plain host pointer on older drivers: clear the sticky error
135 return false;
136 }
137 return a.type == cudaMemoryTypeDevice || a.type == cudaMemoryTypeManaged;
138 }

168 // A matrix as cuBLAS reads it: a column-major array and its leading dimension, transposed if the view is row-major
169 template <typename T>
170 struct operand {
171 const T* data = nullptr;
172 int ld = 1;
173 bool row_major = false; // the column-major array is the view's transpose
174 std::unique_ptr<device_bytes> owned{};
175 };
176
177 // Whether a view is row- or column-major as it stands, and its leading dimension
178 template <typename V>
179 bool blas_layout(const V& v, bool& row_major, std::size_t& ld)
180 {
181 const std::size_t r = v.extent(0), c = v.extent(1);
182 if (v.stride(1) == 1 && (r <= 1 || v.stride(0) >= c)) {
183 row_major = true;
184 ld = std::max<std::size_t>({std::size_t{1}, c, r <= 1 ? c : v.stride(0)});
185 return true;
186 }
187 if (v.stride(0) == 1 && (c <= 1 || v.stride(1) >= r)) {
188 row_major = false;
189 ld = std::max<std::size_t>({std::size_t{1}, r, c <= 1 ? r : v.stride(1)});
190 return true;
191 }
192 return false;
193 }
194
195 // A view's elements, row-major and contiguous, read from wherever they live
196 template <typename T>
197 std::vector<T> packed(linalg::matrix_view<const T> v)
198 {
199 const std::size_t r = v.extent(0), c = v.extent(1), span = v.mapping().required_span_size();
200 std::vector<T> whole(span), out(r * c);
201 if (span != 0)
202 check(cudaMemcpy(whole.data(), v.data_handle(), span * sizeof(T), cudaMemcpyDefault), "cudaMemcpy");
203 for (std::size_t i = 0; i < r; ++i)
204 for (std::size_t j = 0; j < c; ++j)
205 out[i * c + j] = whole[i * v.stride(0) + j * v.stride(1)];
206 return out;
207 }
208
209 // An input on the device: in place when it is device memory in a BLAS layout; host memory in a BLAS layout is
210 // copied up as it is (its span, one transfer); any other strides, packed into a contiguous row-major copy
211 template <typename T>
212 operand<T> input(linalg::matrix_view<const T> v)
213 {
214 operand<T> o;
215 std::size_t ld = 0;
216 if (blas_layout(v, o.row_major, ld)) {
217 o.ld = static_cast<int>(ld);
218 if (on_device(v.data_handle())) {
219 o.data = v.data_handle();
220 return o;
221 }
222 const std::size_t span = v.mapping().required_span_size();
223 o.owned = std::make_unique<device_bytes>(span * sizeof(T));
224 if (span != 0)
225 check(cudaMemcpy(o.owned->p, v.data_handle(), span * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
226 o.data = static_cast<const T*>(o.owned->p);
227 return o;
228 }
229 const auto p = packed(v);
230 o.owned = std::make_unique<device_bytes>(p.size() * sizeof(T));
231 if (!p.empty())
232 check(cudaMemcpy(o.owned->p, p.data(), p.size() * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
233 o.data = static_cast<const T*>(o.owned->p);
234 o.row_major = true;
235 o.ld = static_cast<int>(std::max<std::size_t>(1, v.extent(1)));
236 return o;
237 }

257 [[nodiscard]] static result<std::unique_ptr<cuda_context>, cuda_failure> create(int device = 0)
258 {
259 int count = 0;
260 if (cudaGetDeviceCount(&count) != cudaSuccess || device < 0 || device >= count) {
261 (void)cudaGetLastError();
262 return err(cuda_failure{cuda_error::no_device, 0});
263 }
264 std::unique_ptr<cuda_context> c(new cuda_context(device));
265 if (const cudaError_t e = cudaSetDevice(device); e != cudaSuccess)
266 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
267 if (const cudaError_t e = cudaStreamCreate(&c->stream_); e != cudaSuccess)
268 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
269 if (const cublasStatus_t s = cublasCreate(&c->blas_); s != CUBLAS_STATUS_SUCCESS)
270 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(s)});
271 if (const cublasStatus_t s = cublasSetStream(c->blas_, c->stream_); s != CUBLAS_STATUS_SUCCESS)
272 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(s)});
273 return ok(std::move(c));
274 }

300 template <typename T>
301 requires(std::is_same_v<T, float> || std::is_same_v<T, double>)
302 void gemm(T alpha, linalg::matrix_view<const T> a, linalg::matrix_view<const T> b, T beta, linalg::matrix_view<T> c,
303 cuda_math math = cuda_math::float32) const
304 {
305 namespace dc = detail::cuda;
306 const std::size_t m = c.extent(0), n = c.extent(1), k = a.extent(1);
307 if (m == 0 || n == 0)
308 return;
309 const std::lock_guard lock(mutex_);
310 dc::check(cudaSetDevice(device_), "cudaSetDevice");
311 if (const cublasStatus_t s = cublasSetMathMode(blas_, math == cuda_math::tf32 ? CUBLAS_TF32_TENSOR_OP_MATH : CUBLAS_DEFAULT_MATH);
312 s != CUBLAS_STATUS_SUCCESS)
313 throw cuda_exception("cublasSetMathMode", static_cast<int>(s));
314 const auto A = dc::input(a), B = dc::input(b);
315 // The output: in place when it is device memory in a BLAS layout; host memory in a BLAS layout, its span copied
316 // (up only when it is read or has gaps to keep) and back; any other strides, a contiguous row-major device copy
317 bool c_row = false;
318 std::size_t c_ld = 0;
319 const bool blas_c = dc::blas_layout(c, c_row, c_ld);
320 const bool in_place = blas_c && dc::on_device(c.data_handle());
321 const bool span_copy = blas_c && !in_place;
322 const std::size_t c_span = c.mapping().required_span_size();
323 std::unique_ptr<dc::device_bytes> staged;
324 T* cp = c.data_handle();
325 if (span_copy) {
326 staged = std::make_unique<dc::device_bytes>(c_span * sizeof(T));
327 cp = static_cast<T*>(staged->p);
328 if (beta != T(0) || !c.mapping().is_exhaustive())
329 dc::check(cudaMemcpy(cp, c.data_handle(), c_span * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
330 } else if (!in_place) {
331 staged = std::make_unique<dc::device_bytes>(m * n * sizeof(T));
332 cp = static_cast<T*>(staged->p);
333 c_row = true;
334 c_ld = n;
335 if (beta != T(0)) {
336 const auto p = dc::packed(linalg::matrix_view<const T>(c));
337 dc::check(cudaMemcpy(cp, p.data(), p.size() * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
338 }
339 }
340 const int ik = static_cast<int>(k);
341 auto op = [](bool row_major, bool want_transposed) {
342 return row_major != want_transposed ? CUBLAS_OP_T : CUBLAS_OP_N;
343 };
344 if (!c_row) // C column-major: C = alpha op(A) op(B) + beta C
345 dc::check(dc::gemm(blas_, op(A.row_major, false), op(B.row_major, false), static_cast<int>(m), static_cast<int>(n), ik, &alpha,
346 A.data, A.ld, B.data, B.ld, &beta, cp, static_cast<int>(c_ld)),
347 "cublas gemm");
348 else // C row-major: its column-major array is Cᵀ = alpha Bᵀ Aᵀ + beta Cᵀ
349 dc::check(dc::gemm(blas_, op(B.row_major, true), op(A.row_major, true), static_cast<int>(n), static_cast<int>(m), ik, &alpha,
350 B.data, B.ld, A.data, A.ld, &beta, cp, static_cast<int>(c_ld)),
351 "cublas gemm");
352 // Wait only for a staged copy: it is copied back or freed below. Device memory in place stays queued on the
353 // stream, so the device runs the next step while the host enqueues it
354 if (staged || A.owned || B.owned)
355 dc::check(cudaStreamSynchronize(stream_), "cudaStreamSynchronize");
356 if (span_copy) {
357 dc::check(cudaMemcpy(c.data_handle(), cp, c_span * sizeof(T), cudaMemcpyDeviceToHost), "cudaMemcpy");
358 } else if (!in_place) { // back into the view, wherever it lives, keeping the gaps of its span
359 std::vector<T> result(m * n);
360 dc::check(cudaMemcpy(result.data(), cp, result.size() * sizeof(T), cudaMemcpyDeviceToHost), "cudaMemcpy");
361 const std::size_t span = c.mapping().required_span_size();
362 std::vector<T> whole(span);
363 dc::check(cudaMemcpy(whole.data(), c.data_handle(), span * sizeof(T), cudaMemcpyDefault), "cudaMemcpy");
364 for (std::size_t i = 0; i < m; ++i)
365 for (std::size_t j = 0; j < n; ++j)
366 whole[i * c.stride(0) + j * c.stride(1)] = result[i * n + j];
367 dc::check(cudaMemcpy(c.data_handle(), whole.data(), span * sizeof(T), cudaMemcpyDefault), "cudaMemcpy");
368 }
369 }
