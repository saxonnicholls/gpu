72public:
75 [[nodiscard]] static result<std::unique_ptr<cuda_rows_kernels>, cuda_failure> create(const cuda_context& context)
76 {
77 int major = 0, minor = 0;
78 cudaDeviceGetAttribute(&major, cudaDevAttrComputeCapabilityMajor, context.device());
79 cudaDeviceGetAttribute(&minor, cudaDevAttrComputeCapabilityMinor, context.device());
80 const std::string source = std::string(detail::quant::cuda_prelude) + detail::rows::math_cuda + detail::rows::source;
81 nvrtcProgram prog = nullptr;
82 if (const nvrtcResult r = nvrtcCreateProgram(&prog, source.c_str(), "snicholls_gpu_rows.cu", 0, nullptr, nullptr); r != NVRTC_SUCCESS) {
83 std::fprintf(stderr, "snicholls::gpu::cuda_rows_kernels: nvrtcCreateProgram: %s\n", nvrtcGetErrorString(r));
84 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(r)});
85 }
86 const std::string arch = "--gpu-architecture=sm_" + std::to_string(major * 10 + minor);
87 const char* options[] = {arch.c_str(), "--fmad=false", "--prec-div=true", "--ftz=false", "--std=c++17"};
88 if (const nvrtcResult compiled = nvrtcCompileProgram(prog, 5, options); compiled != NVRTC_SUCCESS) {
89 std::size_t size = 0;
90 nvrtcGetProgramLogSize(prog, &size);
91 std::string log(size, '\0');
92 nvrtcGetProgramLog(prog, log.data());
93 std::fprintf(stderr, "snicholls::gpu::cuda_rows_kernels: %s\n", log.c_str());
94 nvrtcDestroyProgram(&prog);
95 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(compiled)});
96 }
97 std::size_t size = 0;
98 nvrtcGetCUBINSize(prog, &size);
99 std::string cubin(size, '\0');
100 nvrtcGetCUBIN(prog, cubin.data());
101 nvrtcDestroyProgram(&prog);
102 std::unique_ptr<cuda_rows_kernels> k(new cuda_rows_kernels(context));
103 if (const cudaError_t e = cudaLibraryLoadData(&k->library_, cubin.data(), nullptr, nullptr, 0, nullptr, nullptr, 0); e != cudaSuccess) {
104 std::fprintf(stderr, "snicholls::gpu::cuda_rows_kernels: cudaLibraryLoadData: %s\n", cudaGetErrorString(e));
105 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
106 }
107 for (const auto& [name, slot] : {std::pair<const char*, cudaKernel_t*>{"rows_attend", &k->attend_}, {"rows_store", &k->store_}, {"rows_heads", &k->heads_}})
108 if (const cudaError_t e = cudaLibraryGetKernel(slot, k->library_, name); e != cudaSuccess) {
109 std::fprintf(stderr, "snicholls::gpu::cuda_rows_kernels: cudaLibraryGetKernel(%s): %s\n", name, cudaGetErrorString(e));
110 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
111 }
112 return ok(std::move(k));
113 }

127 enum class which {
128 attend,
129 store,
130 heads,
131 };
132
134 void launch(which w, std::size_t groups, unsigned threads, void** args) const
135 {
136 if (groups == 0)
137 return;
138 const cudaKernel_t k = w == which::attend ? attend_ : w == which::store ? store_ : heads_;
139 if (const cudaError_t e = cudaLaunchKernel(reinterpret_cast<const void*>(k), dim3(unsigned(groups)), dim3(threads), args, 0, context_->stream());
140 e != cudaSuccess)
141 throw cuda_exception(std::string("cudaLaunchKernel(rows) [") + cudaGetErrorString(e) + "]", static_cast<int>(e));
142 }
143
144private:
145 explicit cuda_rows_kernels(const cuda_context& c) : context_(&c) {}
146 const cuda_context* context_;
147 cudaLibrary_t library_ = nullptr;
148 cudaKernel_t attend_ = nullptr, store_ = nullptr, heads_ = nullptr;
149};
