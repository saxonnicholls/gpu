137 // The kernels: C = alpha A B + beta C over strided views, 16 x 16 tiles, each output summed in index order with
138 // contraction off, so the result is linalg::reference's
139 inline constexpr const char* gemm_source = R"CLC(
140#pragma OPENCL FP_CONTRACT OFF
141// 64 x 64 tiles of C, 16 deep in k; a work-group of 16 x 16 items, each computing 4 x 4 outputs, every output summed
142// in index order. A tile of an operand is loaded with consecutive items along its unit stride (row-major or
143// column-major), so the loads coalesce whichever way the operand is laid out.
144#define TSM 64
145#define TSN 64
146#define TSK 16
147#define WPT 4
148#define RTS 16
149#define GEMM(NAME, T) \
150__kernel __attribute__((reqd_work_group_size(RTS, RTS, 1))) \
151void NAME(const long m, const long n, const long k, const T alpha, \
152 __global const T* a, const long sa0, const long sa1, __global const T* b, const long sb0, const long sb1, \
153 const T beta, __global T* c, const long sc0, const long sc1) \
154{ \
155 __local T as[TSK][TSM]; \
156 __local T bs[TSK][TSN]; \
157 const int tc = get_local_id(0), tr = get_local_id(1), tid = tr * RTS + tc; \
158 const long col0 = (long)get_group_id(0) * TSN, row0 = (long)get_group_id(1) * TSM; \
159 const bool a_by_rows = sa1 == 1 || sa0 != 1, b_by_rows = sb1 == 1 || sb0 != 1; \
160 T acc[WPT][WPT]; \
161 for (int i = 0; i < WPT; ++i) \
162 for (int j = 0; j < WPT; ++j) \
163 acc[i][j] = (T)0; \
164 for (long t = 0; t < k; t += TSK) { \
165 for (int l = 0; l < (TSM * TSK) / (RTS * RTS); ++l) { \
166 const int e = tid + l * RTS * RTS; \
167 const int ar = a_by_rows ? e / TSK : e % TSM, ap = a_by_rows ? e % TSK : e / TSM; \
168 const long gr = row0 + ar, gp = t + ap; \
169 as[ap][ar] = (gr < m && gp < k) ? a[gr * sa0 + gp * sa1] : (T)0; \
170 const int bp = b_by_rows ? e / TSN : e % TSK, bc = b_by_rows ? e % TSN : e / TSK; \
171 const long gq = t + bp, gc = col0 + bc; \
172 bs[bp][bc] = (gq < k && gc < n) ? b[gq * sb0 + gc * sb1] : (T)0; \
173 } \
174 barrier(CLK_LOCAL_MEM_FENCE); \
175 const int last = k - t < TSK ? (int)(k - t) : TSK; \
176 for (int p = 0; p < last; ++p) { \
177 T ar[WPT], br[WPT]; \
178 for (int i = 0; i < WPT; ++i) \
179 ar[i] = as[p][tr + i * RTS]; \
180 for (int j = 0; j < WPT; ++j) \
181 br[j] = bs[p][tc + j * RTS]; \
182 for (int i = 0; i < WPT; ++i) \
183 for (int j = 0; j < WPT; ++j) \
184 acc[i][j] += ar[i] * br[j]; \
185 } \
186 barrier(CLK_LOCAL_MEM_FENCE); \
187 } \
188 for (int i = 0; i < WPT; ++i) \
189 for (int j = 0; j < WPT; ++j) { \
190 const long row = row0 + tr + i * RTS, col = col0 + tc + j * RTS; \
191 if (row < m && col < n) { \
192 __global T* out = c + row * sc0 + col * sc1; \
193 const T s = acc[i][j]; \
194 *out = beta == (T)0 ? alpha * s : alpha * s + beta * *out; \
195 } \
196 } \
197}
198GEMM(gemm_float, float)
199#ifdef SNICHOLLS_GPU_FP64
200#pragma OPENCL EXTENSION cl_khr_fp64 : enable
201GEMM(gemm_double, double)
202#endif
203)CLC";

214 buffer(cl_context context, cl_mem_flags flags, std::size_t bytes, void* host)
215 {
216 cl_int code = CL_SUCCESS;
217 if (bytes == 0) // an empty span: a one-byte buffer the kernel never reads
218 flags &= ~cl_mem_flags(CL_MEM_COPY_HOST_PTR);
219 mem = clCreateBuffer(context, flags, bytes == 0 ? 1 : bytes, bytes == 0 ? nullptr : host, &code);
220 check(code, "clCreateBuffer");
221 }

242 [[nodiscard]] static result<std::unique_ptr<opencl_context>, opencl_failure> create(device_choice choice = device_choice::any)
243 {
244 cl_uint platforms = 0;
245 if (clGetPlatformIDs(0, nullptr, &platforms) != CL_SUCCESS || platforms == 0)
246 return err(opencl_failure{opencl_error::no_platform, CL_SUCCESS, {}});
247 std::vector<cl_platform_id> ids(platforms);
248 clGetPlatformIDs(platforms, ids.data(), nullptr);
249 cl_device_id device = nullptr;
250 for (const cl_device_type type : {cl_device_type(CL_DEVICE_TYPE_GPU), cl_device_type(CL_DEVICE_TYPE_ALL)}) {
251 if (type == CL_DEVICE_TYPE_ALL && choice == device_choice::gpu)
252 break;
253 for (const cl_platform_id p : ids)
254 if (!device && clGetDeviceIDs(p, type, 1, &device, nullptr) != CL_SUCCESS)
255 device = nullptr;
256 if (device)
257 break;
258 }
259 if (!device)
260 return err(opencl_failure{opencl_error::no_device, CL_SUCCESS, {}});
261 std::unique_ptr<opencl_context> c(new opencl_context(device));
262 cl_int code = CL_SUCCESS;
263 c->context_ = clCreateContext(nullptr, 1, &device, nullptr, nullptr, &code);
264 if (code != CL_SUCCESS)
265 return err(opencl_failure{opencl_error::call_failed, code, {}});
266 c->queue_ = clCreateCommandQueue(c->context_, device, 0, &code);
267 if (code != CL_SUCCESS)
268 return err(opencl_failure{opencl_error::call_failed, code, {}});
269 const char* source = detail::gemm_source; // macOS takes const char**, others const char* const*
270 c->program_ = clCreateProgramWithSource(c->context_, 1, &source, nullptr, &code);
271 if (code != CL_SUCCESS)
272 return err(opencl_failure{opencl_error::call_failed, code, {}});
273 const char* options = c->fp64_ ? "-DSNICHOLLS_GPU_FP64" : "";
274 if (const cl_int built = clBuildProgram(c->program_, 1, &device, options, nullptr, nullptr); built != CL_SUCCESS) {
275 std::size_t size = 0;
276 clGetProgramBuildInfo(c->program_, device, CL_PROGRAM_BUILD_LOG, 0, nullptr, &size);
277 std::string log(size, '\0');
278 clGetProgramBuildInfo(c->program_, device, CL_PROGRAM_BUILD_LOG, size, log.data(), nullptr);
279 return err(opencl_failure{opencl_error::build_failed, built, std::move(log)});
280 }
281 c->gemm_float_ = clCreateKernel(c->program_, "gemm_float", &code);
282 if (code != CL_SUCCESS)
283 return err(opencl_failure{opencl_error::call_failed, code, {}});
284 if (c->fp64_) {
285 c->gemm_double_ = clCreateKernel(c->program_, "gemm_double", &code);
286 if (code != CL_SUCCESS)
287 return err(opencl_failure{opencl_error::call_failed, code, {}});
288 }
289 return ok(std::move(c));
290 }

313 template <typename T>
314 void gemm(T alpha, linalg::matrix_view<const T> a, linalg::matrix_view<const T> b, T beta, linalg::matrix_view<T> c) const
315 {
316 const std::size_t m = c.extent(0), n = c.extent(1), k = a.extent(1);
317 if (m == 0 || n == 0)
318 return;
319 const std::size_t sa = a.mapping().required_span_size(), sb = b.mapping().required_span_size(),
320 sc = c.mapping().required_span_size();
321 const cl_mem_flags in = CL_MEM_READ_ONLY | CL_MEM_COPY_HOST_PTR;
322 // An empty inner dimension still writes alpha * 0 (+ beta c): the kernel's loop runs no times
323 detail::buffer da(context_, in, sa * sizeof(T), const_cast<T*>(a.data_handle()));
324 detail::buffer db(context_, in, sb * sizeof(T), const_cast<T*>(b.data_handle()));
325 // C goes up only if the kernel reads it (beta != 0) or its span has gaps the copy back must keep
326 const bool upload = beta != T(0) || !c.mapping().is_exhaustive();
327 detail::buffer dc(context_, upload ? CL_MEM_READ_WRITE | CL_MEM_COPY_HOST_PTR : CL_MEM_WRITE_ONLY, sc * sizeof(T),
328 upload ? c.data_handle() : nullptr);
329 const cl_long args[] = {cl_long(m), cl_long(n), cl_long(k), cl_long(a.stride(0)), cl_long(a.stride(1)), cl_long(b.stride(0)),
330 cl_long(b.stride(1)), cl_long(c.stride(0)), cl_long(c.stride(1))};
331 const std::lock_guard lock(mutex_);
332 cl_kernel kernel = std::is_same_v<T, float> ? gemm_float_ : gemm_double_;
333 cl_uint i = 0;
334 auto set = [&](std::size_t size, const void* value) { detail::check(clSetKernelArg(kernel, i++, size, value), "clSetKernelArg"); };
335 set(sizeof(cl_long), &args[0]), set(sizeof(cl_long), &args[1]), set(sizeof(cl_long), &args[2]), set(sizeof(T), &alpha);
336 set(sizeof(cl_mem), &da.mem), set(sizeof(cl_long), &args[3]), set(sizeof(cl_long), &args[4]);
337 set(sizeof(cl_mem), &db.mem), set(sizeof(cl_long), &args[5]), set(sizeof(cl_long), &args[6]);
338 set(sizeof(T), &beta), set(sizeof(cl_mem), &dc.mem), set(sizeof(cl_long), &args[7]), set(sizeof(cl_long), &args[8]);
339 const std::size_t global[2] = {(n + 63) / 64 * 16, (m + 63) / 64 * 16}, local[2] = {16, 16}; // 64 x 64 per group
340 detail::check(clEnqueueNDRangeKernel(queue_, kernel, 2, nullptr, global, local, 0, nullptr, nullptr), "clEnqueueNDRangeKernel");
341 detail::check(clEnqueueReadBuffer(queue_, dc.mem, CL_TRUE, 0, sc * sizeof(T), c.data_handle(), 0, nullptr, nullptr),
342 "clEnqueueReadBuffer");
343 }

398 template <linalg::scalar T>

414 template <linalg::scalar T>
415 void gemv(T alpha, linalg::matrix_view<const T> a, linalg::vector_view<const T> x, T beta, linalg::vector_view<T> y) const
416 {
417 if (!on_device<T>()) {
418 linalg::reference.gemv(alpha, a, x, beta, y);
419 return;
420 }
421 context->gemm(alpha, a, linalg::as_column(x), beta, linalg::as_column(y));
422 }

427 template <linalg::scalar T>
428 void gemm(T alpha, linalg::matrix_view<const T> a, linalg::matrix_view<const T> b, T beta, linalg::matrix_view<T> c) const
429 {
430 if (on_device<T>())
431 context->gemm(alpha, a, b, beta, c);
432 else
433 linalg::reference.gemm(alpha, a, b, beta, c);
434 }
435
438 template <linalg::scalar T>
439 void syrk(T alpha, linalg::matrix_view<const T> a, T beta, linalg::matrix_view<T> c, linalg::triangle t) const
440 {
441 linalg::reference.syrk(alpha, a, beta, c, t);
442 }
443
447 template <linalg::scalar T>
448 [[nodiscard]] result<void, linalg::error> triangular_solve(linalg::matrix_view<const T> t, linalg::matrix_view<T> b,
449 linalg::triangle which, linalg::diagonal diag) const
450 {
451 return linalg::reference.triangular_solve(t, b, which, diag);
452 }
453
454private:
455 // float always runs on the device, double when it has cl_khr_fp64, any other scalar on the host
456 template <typename T>
457 [[nodiscard]] bool on_device() const noexcept
458 {
459 return context != nullptr && (std::is_same_v<T, float> || (std::is_same_v<T, double> && context->has_fp64()));
460 }
