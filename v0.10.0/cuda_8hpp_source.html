130 // Whether CUDA knows the pointer as device (or managed) memory
131 inline bool on_device(const void* p) noexcept
132 {
133 if (p == nullptr)
134 return false;
135 cudaPointerAttributes a{};
136 if (cudaPointerGetAttributes(&a, p) != cudaSuccess) {
137 (void)cudaGetLastError(); // a plain host pointer on older drivers: clear the sticky error
138 return false;
139 }
140 return a.type == cudaMemoryTypeDevice || a.type == cudaMemoryTypeManaged;
141 }

174 // A matrix as cuBLAS reads it: a column-major array and its leading dimension, transposed if the view is row-major
175 template <typename T>
176 struct operand {
177 const T* data = nullptr;
178 int ld = 1;
179 bool row_major = false; // the column-major array is the view's transpose
180 std::unique_ptr<device_bytes> owned{};
181 };
182
183 // Whether a view is row- or column-major as it stands, and its leading dimension
184 template <typename V>
185 bool blas_layout(const V& v, bool& row_major, std::size_t& ld)
186 {
187 const std::size_t r = v.extent(0), c = v.extent(1);
188 if (v.stride(1) == 1 && (r <= 1 || v.stride(0) >= c)) {
189 row_major = true;
190 ld = std::max<std::size_t>({std::size_t{1}, c, r <= 1 ? c : v.stride(0)});
191 return true;
192 }
193 if (v.stride(0) == 1 && (c <= 1 || v.stride(1) >= r)) {
194 row_major = false;
195 ld = std::max<std::size_t>({std::size_t{1}, r, c <= 1 ? r : v.stride(1)});
196 return true;
197 }
198 return false;
199 }
200
201 // A view's elements, row-major and contiguous, read from wherever they live
202 template <typename T>
203 std::vector<T> packed(linalg::matrix_view<const T> v)
204 {
205 const std::size_t r = v.extent(0), c = v.extent(1), span = v.mapping().required_span_size();
206 std::vector<T> whole(span), out(r * c);
207 if (span != 0)
208 check(cudaMemcpy(whole.data(), v.data_handle(), span * sizeof(T), cudaMemcpyDefault), "cudaMemcpy");
209 for (std::size_t i = 0; i < r; ++i)
210 for (std::size_t j = 0; j < c; ++j)
211 out[i * c + j] = whole[i * v.stride(0) + j * v.stride(1)];
212 return out;
213 }

215 // An input on the device: in place when it is device memory in a BLAS layout; host memory in a BLAS layout is
216 // copied up as it is (its span, one transfer); any other strides, packed into a contiguous row-major copy
217 template <typename T>
218 operand<T> input(linalg::matrix_view<const T> v)
219 {
220 operand<T> o;
221 std::size_t ld = 0;
222 if (blas_layout(v, o.row_major, ld)) {
223 o.ld = static_cast<int>(ld);
224 if (on_device(v.data_handle())) {
225 o.data = v.data_handle();
226 return o;
227 }
228 const std::size_t span = v.mapping().required_span_size();
229 o.owned = std::make_unique<device_bytes>(span * sizeof(T));
230 if (span != 0)
231 check(cudaMemcpy(o.owned->p, v.data_handle(), span * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
232 o.data = static_cast<const T*>(o.owned->p);
233 return o;
234 }
235 const auto p = packed(v);
236 o.owned = std::make_unique<device_bytes>(p.size() * sizeof(T));
237 if (!p.empty())
238 check(cudaMemcpy(o.owned->p, p.data(), p.size() * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
239 o.data = static_cast<const T*>(o.owned->p);
240 o.row_major = true;
241 o.ld = static_cast<int>(std::max<std::size_t>(1, v.extent(1)));
242 return o;
243 }

267 [[nodiscard]] static result<std::unique_ptr<cuda_context>, cuda_failure> create(int device = 0)
268 {
269 int count = 0;
270 if (cudaGetDeviceCount(&count) != cudaSuccess || device < 0 || device >= count) {
271 (void)cudaGetLastError();
272 return err(cuda_failure{cuda_error::no_device, 0});
273 }
274 std::unique_ptr<cuda_context> c(new cuda_context(device));
275 if (const cudaError_t e = cudaSetDevice(device); e != cudaSuccess)
276 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
277 if (const cudaError_t e = cudaStreamCreate(&c->stream_); e != cudaSuccess)
278 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
279 if (const cublasStatus_t s = cublasCreate(&c->blas_); s != CUBLAS_STATUS_SUCCESS)
280 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(s)});
281 if (const cublasStatus_t s = cublasSetStream(c->blas_, c->stream_); s != CUBLAS_STATUS_SUCCESS)
282 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(s)});
283 return ok(std::move(c));
284 }

310 template <typename T>
311 requires(std::is_same_v<T, float> || std::is_same_v<T, double>)
312 void gemm(T alpha, linalg::matrix_view<const T> a, linalg::matrix_view<const T> b, T beta, linalg::matrix_view<T> c,
313 cuda_math math = cuda_math::float32) const
314 {
315 namespace dc = detail::cuda;
316 const std::size_t m = c.extent(0), n = c.extent(1), k = a.extent(1);
317 if (m == 0 || n == 0)
318 return;
319 const std::lock_guard lock(mutex_);
320 dc::check(cudaSetDevice(device_), "cudaSetDevice");
321 if (const cublasStatus_t s = cublasSetMathMode(blas_, math == cuda_math::tf32 ? CUBLAS_TF32_TENSOR_OP_MATH : CUBLAS_DEFAULT_MATH);
322 s != CUBLAS_STATUS_SUCCESS)
323 throw cuda_exception("cublasSetMathMode", static_cast<int>(s));
324 const auto A = dc::input(a), B = dc::input(b);
325 // The output: in place when it is device memory in a BLAS layout; host memory in a BLAS layout, its span copied
326 // (up only when it is read or has gaps to keep) and back; any other strides, a contiguous row-major device copy
327 bool c_row = false;
328 std::size_t c_ld = 0;
329 const bool blas_c = dc::blas_layout(c, c_row, c_ld);
330 const bool in_place = blas_c && dc::on_device(c.data_handle());
331 const bool span_copy = blas_c && !in_place;
332 const std::size_t c_span = c.mapping().required_span_size();
333 std::unique_ptr<dc::device_bytes> staged;
334 T* cp = c.data_handle();
335 if (span_copy) {
336 staged = std::make_unique<dc::device_bytes>(c_span * sizeof(T));
337 cp = static_cast<T*>(staged->p);
338 if (beta != T(0) || !c.mapping().is_exhaustive())
339 dc::check(cudaMemcpy(cp, c.data_handle(), c_span * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
340 } else if (!in_place) {
341 staged = std::make_unique<dc::device_bytes>(m * n * sizeof(T));
342 cp = static_cast<T*>(staged->p);
343 c_row = true;
344 c_ld = n;
345 if (beta != T(0)) {
346 const auto p = dc::packed(linalg::matrix_view<const T>(c));
347 dc::check(cudaMemcpy(cp, p.data(), p.size() * sizeof(T), cudaMemcpyHostToDevice), "cudaMemcpy");
348 }
349 }
350 const int ik = static_cast<int>(k);
351 auto op = [](bool row_major, bool want_transposed) {
352 return row_major != want_transposed ? CUBLAS_OP_T : CUBLAS_OP_N;
353 };
354 if (!c_row) // C column-major: C = alpha op(A) op(B) + beta C
355 dc::check(dc::gemm(blas_, op(A.row_major, false), op(B.row_major, false), static_cast<int>(m), static_cast<int>(n), ik, &alpha,
356 A.data, A.ld, B.data, B.ld, &beta, cp, static_cast<int>(c_ld), math == cuda_math::bf16),
357 "cublas gemm");
358 else // C row-major: its column-major array is Cᵀ = alpha Bᵀ Aᵀ + beta Cᵀ
359 dc::check(dc::gemm(blas_, op(B.row_major, true), op(A.row_major, true), static_cast<int>(n), static_cast<int>(m), ik, &alpha,
360 B.data, B.ld, A.data, A.ld, &beta, cp, static_cast<int>(c_ld), math == cuda_math::bf16),
361 "cublas gemm");
362 // Wait only for a staged copy: it is copied back or freed below. Device memory in place stays queued on the
363 // stream, so the device runs the next step while the host enqueues it
364 if (staged || A.owned || B.owned)
365 dc::check(cudaStreamSynchronize(stream_), "cudaStreamSynchronize");
366 if (span_copy) {
367 dc::check(cudaMemcpy(c.data_handle(), cp, c_span * sizeof(T), cudaMemcpyDeviceToHost), "cudaMemcpy");
368 } else if (!in_place) { // back into the view, wherever it lives, keeping the gaps of its span
369 std::vector<T> result(m * n);
370 dc::check(cudaMemcpy(result.data(), cp, result.size() * sizeof(T), cudaMemcpyDeviceToHost), "cudaMemcpy");
371 const std::size_t span = c.mapping().required_span_size();
372 std::vector<T> whole(span);
373 dc::check(cudaMemcpy(whole.data(), c.data_handle(), span * sizeof(T), cudaMemcpyDefault), "cudaMemcpy");
374 for (std::size_t i = 0; i < m; ++i)
375 for (std::size_t j = 0; j < n; ++j)
376 whole[i * c.stride(0) + j * c.stride(1)] = result[i * n + j];
377 dc::check(cudaMemcpy(c.data_handle(), whole.data(), span * sizeof(T), cudaMemcpyDefault), "cudaMemcpy");
378 }
379 }

387 void gemm_bf16(float alpha, linalg::matrix_view<const std::uint16_t> a, linalg::matrix_view<const std::uint16_t> b, float beta,
388 linalg::matrix_view<float> c) const
389 {
390 namespace dc = detail::cuda;
391 const std::size_t m = c.extent(0), n = c.extent(1), k = a.extent(1);
392 if (m == 0 || n == 0)
393 return;
394 bool ar = false, br = false, cr = false;
395 std::size_t lda = 0, ldb = 0, ldc = 0;
396 if (!dc::blas_layout(a, ar, lda) || !dc::blas_layout(b, br, ldb) || !dc::blas_layout(c, cr, ldc) || !dc::on_device(a.data_handle())
397 || !dc::on_device(b.data_handle()) || !dc::on_device(c.data_handle()))
398 throw std::invalid_argument("snicholls::gpu::cuda_context::gemm_bf16: every operand must be device memory in a BLAS layout");
399 const std::lock_guard lock(mutex_);
400 dc::check(cudaSetDevice(device_), "cudaSetDevice");
401 auto op = [](bool row_major, bool want_transposed) { return row_major != want_transposed ? CUBLAS_OP_T : CUBLAS_OP_N; };
402 const auto ex = [&](cublasOperation_t ta, cublasOperation_t tb, std::size_t mm, std::size_t nn, const void* x, std::size_t ldx, const void* y,
403 std::size_t ldy) {
404 return cublasGemmEx(blas_, ta, tb, int(mm), int(nn), int(k), &alpha, x, CUDA_R_16BF, int(ldx), y, CUDA_R_16BF, int(ldy), &beta,
405 c.data_handle(), CUDA_R_32F, int(ldc), CUBLAS_COMPUTE_32F, CUBLAS_GEMM_DEFAULT);
406 };
407 if (!cr)
408 dc::check(ex(op(ar, false), op(br, false), m, n, a.data_handle(), lda, b.data_handle(), ldb), "cublas gemm bf16");
409 else // C row-major: Cᵀ = Bᵀ Aᵀ, as in gemm
410 dc::check(ex(op(br, true), op(ar, true), n, m, b.data_handle(), ldb, a.data_handle(), lda), "cublas gemm bf16");
411 }

500class cuda_graph {
501public:
506 template <typename F>
507 [[nodiscard]] static result<cuda_graph, cuda_failure> capture(const cuda_context& context, F&& enqueue)
508 {
509 cudaStream_t s = context.stream();
510 if (const cudaError_t e = cudaStreamBeginCapture(s, cudaStreamCaptureModeThreadLocal); e != cudaSuccess)
511 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
512 cudaGraph_t graph = nullptr;
513 try {
514 std::forward<F>(enqueue)();
515 } catch (...) {
516 (void)cudaStreamEndCapture(s, &graph);
517 if (graph != nullptr)
518 (void)cudaGraphDestroy(graph);
519 (void)cudaGetLastError();
520 throw;
521 }
522 if (const cudaError_t e = cudaStreamEndCapture(s, &graph); e != cudaSuccess) {
523 if (graph != nullptr)
524 (void)cudaGraphDestroy(graph);
525 (void)cudaGetLastError();
526 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
527 }
528 cuda_graph out(context.stream());
529 out.graph_ = graph;
530 if (const cudaError_t e = cudaGraphInstantiate(&out.exec_, graph, 0); e != cudaSuccess)
531 return err(cuda_failure{cuda_error::call_failed, static_cast<int>(e)});
532 return ok(std::move(out));
533 }
534
536 cuda_graph(cuda_graph&& o) noexcept : stream_(o.stream_), graph_(std::exchange(o.graph_, nullptr)), exec_(std::exchange(o.exec_, nullptr)) {}
538 cuda_graph& operator=(cuda_graph&& o) noexcept
539 {
540 if (this != &o) {
541 release();
542 stream_ = o.stream_, graph_ = std::exchange(o.graph_, nullptr), exec_ = std::exchange(o.exec_, nullptr);
543 }
544 return *this;
545 }
546 cuda_graph(const cuda_graph&) = delete;
547 cuda_graph& operator=(const cuda_graph&) = delete;
549 ~cuda_graph() { release(); }
550
552 void launch() const { detail::cuda::check(cudaGraphLaunch(exec_, stream_), "cudaGraphLaunch"); }
553
555 [[nodiscard]] std::size_t nodes() const
556 {
557 std::size_t n = 0;
558 detail::cuda::check(cudaGraphGetNodes(graph_, nullptr, &n), "cudaGraphGetNodes");
559 return n;
560 }
561
562private:
563 explicit cuda_graph(cudaStream_t s) noexcept : stream_(s) {}
564 void release() noexcept
565 {
566 if (exec_ != nullptr)
567 (void)cudaGraphExecDestroy(exec_);
568 if (graph_ != nullptr)
569 (void)cudaGraphDestroy(graph_);
570 exec_ = nullptr, graph_ = nullptr;
571 }
572 cudaStream_t stream_ = nullptr;
573 cudaGraph_t graph_ = nullptr;
574 cudaGraphExec_t exec_ = nullptr;
575};
